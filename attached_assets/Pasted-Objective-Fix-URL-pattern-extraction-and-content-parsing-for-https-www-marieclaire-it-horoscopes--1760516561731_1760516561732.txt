def extract_sign_content(h2_element):
    content = []
    next_sib = h2_element.next_sibling
    while next_sib and next_sib.name != 'h2':
        if next_sib.name == 'p':
            content.append(next_sib.text)
        next_sib = next_sib.next_sibling
    return " ".join(content)
